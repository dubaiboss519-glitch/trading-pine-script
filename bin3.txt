//@version=5
indicator("Fractal Trend Binary Strategy", overlay=true)

// ===== INPUTS =====
emaLen = input.int(100, "Trend EMA")
expiry = input.int(1, "Expiry")
payout = input.float(0.8)

startBal = input.float(100)
tradeSize = input.float(1)

// ===== TREND =====
ema = ta.ema(close, emaLen)
upTrend   = close > ema
downTrend = close < ema

// ===== FRACTALS =====
fractalUp = ta.highest(high, 3)[1] == high[1]   // top fractal
fractalDown = ta.lowest(low, 3)[1] == low[1]    // bottom fractal

// ===== SIGNAL LOGIC =====
callRaw = fractalDown and upTrend
putRaw  = fractalUp and downTrend

// ===== NEXT CANDLE =====
callSignal = callRaw[1]
putSignal  = putRaw[1]

// ===== STATE =====
var float balance = startBal
var int wins = 0
var int losses = 0

var float entryPrice = na
var int entryBar = na
var int direction = 0

// ===== ENTRY =====
if callSignal
    entryPrice := close
    entryBar := bar_index
    direction := 1

if putSignal
    entryPrice := close
    entryBar := bar_index
    direction := -1

// ===== RESULT =====
win = false
loss = false

if not na(entryBar) and bar_index == entryBar + expiry
    if direction == 1
        win := close > entryPrice
        loss := close <= entryPrice
    if direction == -1
        win := close < entryPrice
        loss := close >= entryPrice

    if win
        wins += 1
        balance += tradeSize * payout
    if loss
        losses += 1
        balance -= tradeSize

    entryBar := na

// ===== STATS =====
total = wins + losses
wr = total > 0 ? (wins / total) * 100 : 0

// ===== PLOTS =====
plot(ema, color=color.yellow)

plotshape(callSignal, location=location.belowbar, color=color.lime, text="BUY")
plotshape(putSignal, location=location.abovebar, color=color.red, text="SELL")

plotshape(win, location=location.belowbar, color=color.blue, style=shape.circle)
plotshape(loss, location=location.abovebar, color=color.orange, style=shape.cross)

// ===== DASHBOARD =====
var label stats = na
if bar_index % 10 == 0
    label.delete(stats)
    stats := label.new(bar_index, high,
     "Balance: $" + str.tostring(balance, "#.##") +
     "\nWinrate: " + str.tostring(wr, "#.##") + "%",
     textcolor=color.white)
